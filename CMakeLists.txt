cmake_minimum_required(VERSION 3.25)
project(Ambergris VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AMBER_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

include(CTest)
include(cmake/Dependencies.cmake)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(AMBER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(server/store)
add_subdirectory(server/record)
add_subdirectory(server/snap)
add_subdirectory(server/step)
add_subdirectory(server/react)
add_subdirectory(server/catalog)
add_subdirectory(server/install)
add_subdirectory(server/lifecycle)
add_subdirectory(server/command)
add_subdirectory(server/perm)
add_subdirectory(server/bindings/execution)
add_subdirectory(server/drive)
add_subdirectory(server/space)
add_subdirectory(server/observe)
add_subdirectory(server/net)
add_subdirectory(server/control)
add_subdirectory(server/elaborate)
add_subdirectory(server/bindings)

add_executable(amb-server server/main.cpp server/settings.cpp server/recovery.cpp)
target_link_libraries(amb-server PRIVATE amber_bindings
  Ambergris::tomlplusplus Ambergris::TracyClient)

add_executable(foundation_bench bench/foundation.cpp)
target_link_libraries(foundation_bench PRIVATE amber_record)
add_executable(permissions_bench bench/permissions.cpp)
target_link_libraries(permissions_bench PRIVATE amber_perm Ambergris::nlohmann_json)
add_executable(grant_deadlines_bench bench/grant_deadlines.cpp)
target_link_libraries(grant_deadlines_bench PRIVATE amber_perm Ambergris::nlohmann_json)
add_executable(snapshot_bench bench/snapshot.cpp)
target_link_libraries(snapshot_bench PRIVATE amber_snap Ambergris::nlohmann_json)

if(BUILD_TESTING)
  add_executable(recovery_test tests/recovery_test.cpp server/recovery.cpp)
  target_include_directories(recovery_test PRIVATE server)
  add_test(NAME recovery COMMAND recovery_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME architecture COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/check_architecture.py)
endif()
