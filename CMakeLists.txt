cmake_minimum_required(VERSION 3.25)
project(Ambergris VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(AMBER_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

include(CTest)
include(cmake/Dependencies.cmake)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(AMBER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(server/store)
add_subdirectory(server/record)
add_subdirectory(server/snap)
add_subdirectory(server/react)
add_subdirectory(server/perm)
add_subdirectory(server/space)
add_subdirectory(server/net)
add_subdirectory(server/elaborate)
add_subdirectory(server/step)

add_executable(amb-server server/main.cpp server/settings.cpp server/recovery.cpp)
target_link_libraries(amb-server PRIVATE amber_step Ambergris::tomlplusplus Ambergris::TracyClient)

include(cmake/Tests.cmake)
